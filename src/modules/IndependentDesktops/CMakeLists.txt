cmake_minimum_required(VERSION 3.20)
project(IndependentDesktops LANGUAGES CXX)
enable_testing()
add_executable(IndependentDesktopsCoreTests Tests/CoreTests.cpp)
target_compile_features(IndependentDesktopsCoreTests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(IndependentDesktopsCoreTests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(IndependentDesktopsCoreTests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME IndependentDesktopsCore COMMAND IndependentDesktopsCoreTests)
if(WIN32)
  enable_language(RC)
  add_executable(IndependentDesktops WIN32 main.cpp Recovery.cpp WindowOperations.cpp IndependentDesktops.rc app.manifest)
  set_target_properties(IndependentDesktops PROPERTIES OUTPUT_NAME PowerToys.IndependentDesktops)
  target_compile_features(IndependentDesktops PRIVATE cxx_std_20)
  target_compile_definitions(IndependentDesktops PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN INDEPENDENT_DESKTOPS_STANDALONE)
  target_compile_options(IndependentDesktops PRIVATE /W4 /WX /permissive-)
  target_link_libraries(IndependentDesktops PRIVATE user32 advapi32 ole32 dwmapi shell32)
  add_executable(IndependentDesktopsRuntimeTests Tests/RuntimeTests.cpp Recovery.cpp WindowOperations.cpp)
  target_compile_features(IndependentDesktopsRuntimeTests PRIVATE cxx_std_20)
  target_compile_definitions(IndependentDesktopsRuntimeTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(IndependentDesktopsRuntimeTests PRIVATE /W4 /WX /permissive-)
  target_link_libraries(IndependentDesktopsRuntimeTests PRIVATE user32 advapi32)
  add_test(NAME IndependentDesktopsRuntime COMMAND IndependentDesktopsRuntimeTests)
endif()
